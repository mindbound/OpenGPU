# Tier-E oracle goldens (28 §11.4; S6 plan §6.4 with amendments C and D). Own work, MIT.
# Every grid is derived by hand from 28 §3.4-§3.5, §5.1-§5.7, §6.1-§6.4 and the synthetic 6x13 glyph rule
# of 28:633 (U+0020 and U+00A0 blank; cp in U+0021-007E, U+00A1-00FF: row r = ((7cp + 29r) mod 256) AND 0xFC;
# anything else the box, rows 0 and 12 = 0xFC, rows 1-11 = 0x84; column b is bit 7 - b), never by the Java
# oracle, and re-derived by oracle_goldens.py. Each case is a new device: handles start at 0x08000000 and
# come from a FIFO free queue (28 §6.1), so a freed slot goes to the tail and the next create takes slot n+1.
# A glyph's six columns are written below as a 6-digit binary (b0 first), e.g. 'A' r0 = 0xC4 -> 110001.
glyphs synthetic-6x13 28:633

case fill-rgb565-quantize

# blit at (9,5) writes only the 07 texels (10,5) and (9,6); a transparent 'o' (rows as in
# text-transparent-over-fill) at x 1-6 writes 09 over both regions; a fill at (8,10) 4x5 clips at the bottom
surface index8 12x13
chars .=01 #=05 z=00 k=07 *=09 f=06
frame
  segment
    clear 1
    fill 0 6 12 7 5
totals records=2 clear=156 fill=84 copy=0 set=0
expect
  ............
  ............
  ............
  ............
  ............
  ............
  ############
  ############
  ############
  ############
  ############
  ############
  ############
texture 0x08000000 2x2
  zk
  kz
frame
  segment
    blit 0x08000000 0 0 2 2 9 5 1
    text 1 0 9 2 1 0 "o"
    fill 8 10 4 5 6
totals records=3 clear=0 fill=12 copy=4 set=78
expect
  .....*......
  ...*..*.....
  ..*.........
  ..**........
  ..*****.....
  .*..**....k.
  #*#**#*##k##
  #**#*#*#####
  #****#######
  #####**#####
  ###*#*##ffff
  ##*##*##ffff
  ##**##*#ffff
end

case frame-two-segments
# one FRAME of two segments: segment 2's fill (04) lands after segment 1's clear (02) and fill (03); a segment
# is not a scope (28 §3.2), and the totals sum both segments
surface index8 6x4
chars .=02 a=03 b=04
frame
  segment
    clear 2
    fill 0 0 3 4 3
  segment
    fill 2 1 3 2 4
totals records=3 clear=24 fill=18 copy=0 set=0
expect
  aaa...
  aabbb.
  aabbb.
  aaa...
end

case frame-with-empty-segment
# a 9-byte segment (header and END) is a no-op with no records (28 §3.1, §3.5); the frame's other segment
# fills x 1-2, rows 0-1
surface index8 4x3
frame
  segment
  segment
    fill 1 0 2 2 0x0C
totals records=1 clear=0 fill=4 copy=0 set=0
expect
  00 0C 0C 00
  00 0C 0C 00
  00 00 00 00
end

case update-subrect-then-blit
# a 4x4 texture, then TEX_UPDATEs of the 3x1 region at (1,2) and the 1x2 region at (0,0), rows top first
# (28 §6.3); the blit to (1,1) shows both updates and the untouched texels
surface index8 6x6
texture 0x08000000 4x4
  11 12 13 14
  21 22 23 24
  31 32 33 34
  41 42 43 44
update 0x08000000 1 2 3x1
  AA BB CC
update 0x08000000 0 0 1x2
  D1
  D2
frame
  segment
    blit 0x08000000 0 0 4 4 1 1 0
totals records=1 clear=0 fill=0 copy=16 set=0
expect
  00 00 00 00 00 00
  00 D1 12 13 14 00
  00 D2 22 23 24 00
  00 31 AA BB CC 00
  00 41 42 43 44 00
  00 00 00 00 00 00
end

case update-subrect-then-blit-rgb565
# a zeroed 3x2 rgb565 texture (28 §6.2), a 2x2 update at (1,0) with LE u16 payload rows, then a blit with
# flags 0 to (0,1): texel column 0 stays 0x0000 and is written over FFFF
surface rgb565 4x3
texture 0x08000000 3x2 zeros
update 0x08000000 1 0 2x2
  1234 ABCD
  F800 07E0
frame
  segment
    clear 0xFFFFFFFF
    blit 0x08000000 0 0 3 2 0 1 0
totals records=2 clear=12 fill=0 copy=6 set=0
expect
  FFFF FFFF FFFF FFFF
  0000 1234 ABCD FFFF
  0000 F800 07E0 FFFF
end

case free-then-new-handle
# free puts slot 0 at the FIFO tail with generation 1 (28 §6.1), so the next create takes slot 1, 0x08010000;
# frame 2 blits the new texture and frame 1's pixels carry over
surface index8 4x2
texture 0x08000000 2x1
  11 12
frame
  segment
    blit 0x08000000 0 0 2 1 0 0 0
totals records=1 clear=0 fill=0 copy=2 set=0
expect
  11 12 00 00
  00 00 00 00
free 0x08000000
texture 0x08010000 2x1
  21 22
frame
  segment
    blit 0x08010000 0 0 2 1 2 1 0
totals records=1 clear=0 fill=0 copy=2 set=0
expect
  11 12 00 00
  00 00 21 22
end

case reset-then-frame
# RESET replaces the surface with its pixels and drops every texture (28:496); the reset frees 0x08000000
# (generation 1, to the queue's tail), so the next create is slot 1, 0x08010000; the frame after it changes
# only (0,0) and (3,2), and the rest shows the reset's pixels
surface index8 4x3
texture 0x08000000 2x2
  11 12
  21 22
frame
  segment
    clear 9
    blit 0x08000000 0 0 2 2 0 0 0
totals records=2 clear=12 fill=0 copy=4 set=0
expect
  11 12 09 09
  21 22 09 09
  09 09 09 09
reset
  01 02 03 04
  05 06 07 08
  0A 0B 0C 0D
texture 0x08010000 1x1
  EE
frame
  segment
    blit 0x08010000 0 0 1 1 3 2 0
    fill 0 0 1 1 0x30
totals records=2 clear=0 fill=1 copy=1 set=0
expect
  30 02 03 04
  05 06 07 08
  0A 0B 0C EE
end
